## **Работа с деревом:**

```c
// Лексикографическое сравнение двух контактов для определения направления в дереве 
// Ключ дерева: сначала фамилия, при равенстве - имя
int compareContacts(Contact *a, Contact *b) {
    // Сравниваем фамилии как строки
    int cmpLast = strcmp(a->lastName, b->lastName);
    // Если фамилии различаются - порядок уже определён
    if (cmpLast != 0) return cmpLast;
    // Фамилии совпали - сравниваем имена
    return strcmp(a->firstName, b->firstName);
}


// Рекурсивная вставка контакта в бинарное дерево поиска
void insertNode(Contact val, Item **q) {
    // Если текущая позиция пуста => создаём новый лист
    if (*q == NULL) {
        // Выделяем память под новый узел
        *q = malloc(sizeof(Item));
        // У нового узла пока нет потомков
        (*q)->left = (*q)->right = NULL;
        // Копируем данные контакта в узел
        (*q)->contact = val;
        // Вставка завершена
        return;
    }
    // Если ключ текущего узла лексикографически больше вставляемого — идём влево
    if (compareContacts(&(*q)->contact, &val) > 0)
        insertNode(val, &(*q)->left);
    // Иначе (меньше или равен) => идём вправо
    else
        insertNode(val, &(*q)->right);
}


// Рекурсивное удаление узла из дерева по ключу (фамилия + имя)
int deleteNode(Contact key, Item **node) {
    Item *t, *up; // t - временный узел; up - «родитель» t при поиске преемника

    // Ключа в этом поддереве нет
    if (*node == NULL) return 0;

    // Сравниваем ключ текущего узла с искомым
    int cmp = compareContacts(&(*node)->contact, &key);

    // Нашли узел с нужным ключом
    if (cmp == 0) {
        // 1) узел лист (нет детей)
        if (((*node)->left == NULL) && ((*node)->right == NULL)) {
            free(*node);   // Освобождаем память
            *node = NULL;  // Родитель теряет ссылку на узел
            return 1;      // Удаление успешно
        }
        // 2) только правый потомок
        if ((*node)->left == NULL) {
            t = *node;                    // Запоминаем удаляемый узел
            *node = (*node)->right;       // Поднимаем правого ребёнка на место удаляемого
            free(t);                      // Освобождаем старый узел
            return 1;
        }
        // 3) только левый потомок
        if ((*node)->right == NULL) {
            t = *node;
            *node = (*node)->left;        // Поднимаем левого ребёнка
            free(t);
            return 1;
        }
        // 4) два потомка => ищем максимум в левом поддереве
        up = *node;                       // up изначально указывает на удаляемый узел
        t = (*node)->left;                // Идём в левое поддерево

        // Спускаемся до крайнего правого узла левого поддерева
        while (t->right != NULL) {
            up = t;                       // up - родитель t на пути вниз
            t = t->right;
        }

        // Копируем данные преемника в текущий узел (структура узла сохраняется)
        (*node)->contact = t->contact;

        // Преемник не является прямым левым ребёнком удаляемого узла
        if (up != (*node)) {
            // Поднимаем левого ребёнка преемника (если есть) на место преемника
            if (t->left != NULL) up->right = t->left;
            else up->right = NULL;        // Преемник был листом справа у up
        }
        // Преемник - прямой левый ребёнок удаляемого узла
        else {
            (*node)->left = t->left;      // Левый ребёнок преемника становится левым ребёнком узла
        }

        free(t);                          // Удаляем узел-преемник (данные уже скопированы)
        return 1;
    }

    // Ключ меньше текущего => ищем в левом поддереве
    if (cmp > 0)
        return deleteNode(key, &(*node)->left);
    // Ключ больше => ищем в правом поддереве
    return deleteNode(key, &(*node)->right);
}


// Симметричный (in-order) обход дерева для вывода таблицы контактов
void printInOrder(Item *p, int *idx) {
    // База рекурсии: пустое поддерево
    if (p == NULL) return;
    // Сначала обходим левое поддерево (контакты с меньшими ключами)
    printInOrder(p->left, idx);
    // Выводим текущий узел с порядковым номером для пользователя
    printf("| %-3d | %-19s | %-19s |\n", *idx, p->contact.lastName, p->contact.firstName);
    // Увеличиваем счётчик отображаемого номера
    (*idx)++;
    // Затем обходим правое поддерево (контакты с большими ключами)
    printInOrder(p->right, idx);
}


// In-order обход: заполняет массив указателей на узлы в отсортированном порядке
void flattenToArray(Item *p, Item **arr, int *currentIndex) {
    // Пустое поддерево - ничего не делаем
    if (p == NULL) return;
    // Обходим левое поддерево
    flattenToArray(p->left, arr, currentIndex);
    // Кладём указатель на текущий узел в следующую ячейку массива
    arr[*currentIndex] = p;
    // Сдвигаем индекс заполнения массива
    (*currentIndex)++;
    // Обходим правое поддерево
    flattenToArray(p->right, arr, currentIndex);
}


// Рекурсивное построение сбалансированного дерева из отсортированного массива указателей
Item* buildBalanced(Item **arr, int start, int end) {
    // Нет элементов в диапазоне — пустое поддерево
    if (start > end) return NULL;
    // Середина диапазона - корень поддерева (минимальная высота)
    int mid = (start + end) / 2;
    // Берём существующий узел из массива (новый узел не создаётся)
    Item *node = arr[mid];
    // Левое поддерево строится из левой половины массива
    node->left = buildBalanced(arr, start, mid - 1);
    // Правое поддерево из правой половины
    node->right = buildBalanced(arr, mid + 1, end);
    // Возвращаем корень построенного поддерева
    return node;
}


// Перестраивает дерево в сбалансированное без изменения набора контактов
void balanceTree() {
    // Нечего балансировать
    if (contactCount == 0) return;

    // Массив указателей на все узлы (размер = число контактов)
    Item **arr = malloc(contactCount * sizeof(Item*));
    // Индекс заполнения массива, начинаем с 0
    int current = 0;
    // In-order обход: заполняем arr отсортированными указателями на узлы
    flattenToArray(root, arr, &current);

    // Пересобираем связи left/right
    // root становится корнем нового сбалансированного дерева
    root = buildBalanced(arr, 0, contactCount - 1);
    // Массив указателей больше не нужен (сами узлы остаются в памяти)
    free(arr);

    // Сбрасываем счётчик операций до следующей автобалансировки
    operationsSinceBalance = 0;
    printf("\n[Система]: Дерево успешно сбалансировано.\n");
}


// Рекурсивное освобождение памяти всего дерева (post-order обход)
void freeTree(Item *p) {
    // Пустое поддерево — нечего освобождать
    if (p == NULL) return;
    // Сначала освобождаем левое поддерево
    freeTree(p->left);
    // Затем правое поддерево
    freeTree(p->right);
    // После детей - сам узел
    free(p);
}


// Поиск элемента по визуальному индексу (0..contactCount-1) через in-order обход
Item* getItemByIndex(int index) {
    // Индекс вне допустимого диапазона
    if (index < 0 || index >= contactCount) return NULL;

    // Временный массив указателей на все узлы в отсортированном порядке
    Item **arr = malloc(contactCount * sizeof(Item*));
    int current = 0;
    // Заполняем массив in-order обходом
    flattenToArray(root, arr, &current);

    // Берём узел по индексу
    Item *res = arr[index];
    // Освобождаем только массив указателей, не сами узлы
    free(arr);
    return res;
}


// Полное освобождение памяти дерева и сброс глобального состояния
void freeContacts() {
    freeTree(root);     // Рекурсивно освобождаем все узлы
    root = NULL;        // Дерево пусто
    contactCount = 0;   // Сбрасываем счётчик контактов
}
```

```c
// Функция для вывода главного меню

// Создание паузы для ожидания нажатия Enter
void waitEnter() {
    printf("\nНажмите Enter, чтобы продолжить...");
    char temp[10];
    fgets(temp, sizeof(temp), stdin);
}

// Безопасное чтение строки с клавиатуры
void getInput(char *buffer, int size) {
    fgets(buffer, size, stdin);
    buffer[strcspn(buffer, "\n")] = '\0'; // Убираем символ перевода строки в конце
}

// Защищённый ввод числа с проверкой диапазона
int validateNumberInput(int min, int max) {
    int input;
    char buffer[100]; // Временный текстовый буфер для считывания ввода
    while (1) {
        // Читаем весь ввод пользователя как обычный текст
        if (!fgets(buffer, sizeof(buffer), stdin)) {
            printf("Ошибка чтения ввода!\n");
            continue;
        }
        // Пытаемся вытащить число из считанного текста
        if (sscanf(buffer, "%d", &input) != 1) {
            printf("Ошибка: Введите число от %d до %d: ", min, max);
            continue;
        }
        // Проверяем, попадает ли введённое число в заданный диапазон
        if (input < min || input > max) {
            printf("Ошибка: Число должно быть от %d до %d: ", min, max);
            continue;
        }
        return input;
    }
}

// Быстрая проверка на то, не пуста ли телефонная книга
bool checkContactsExist() {

// Функция добавления нового контакта: запрашивает данные и вставляет узел в дерево
void addContact() {
    Contact newContact = {0}; // Создаём пустой контакт (заполнен нулями)
    int validLast = 0, validFirst = 0;

    printf("\n--- ДОБАВЛЕНИЕ КОНТАКТА ---\n");

    // Обязательные поля запрашиваем в цикле, пока пользователь их не введёт
    do {
        printf("Введите фамилию: ");
        getInput(newContact.lastName, MAX_LENGTH);
        validLast = strlen(newContact.lastName) > 0;
        if (!validLast) printf("Ошибка: Фамилия обязательна!\n");

        printf("Введите имя: ");
        getInput(newContact.firstName, MAX_LENGTH);
        validFirst = strlen(newContact.firstName) > 0;
        if (!validFirst) printf("Ошибка: Имя обязательно!\n");
    } while (!validLast || !validFirst);

    // Дополнительные поля (опционально)
    printf("Место работы (Enter - пропустить): ");
    getInput(newContact.workplace, MAX_LENGTH);
    printf("Должность (Enter - пропустить): ");
    getInput(newContact.job, MAX_LENGTH);
    printf("Телефон (Enter - пропустить): ");
    getInput(newContact.phone, MAX_LENGTH);
    printf("Email (Enter - пропустить): ");
    getInput(newContact.email, MAX_LENGTH);
    printf("Соцсети (Enter - пропустить): ");
    getInput(newContact.social, MAX_LENGTH);
    printf("Мессенджер (Enter - пропустить): ");
    getInput(newContact.messenger, MAX_LENGTH);

    // Вставляем контакт в дерево поиска (память выделяется внутри insertNode)
    insertNode(newContact, &root);
    contactCount++;              // Увеличиваем счётчик контактов
    operationsSinceBalance++;    // Учитываем операцию для автобалансировки

    printf("\nКонтакт успешно добавлен!\n");

    // Периодическая балансировка после 5 операций add/delete
    if (operationsSinceBalance >= 5) balanceTree();

    waitEnter();
}

// Функция вывода списка контактов (только фамилии и имена с нумерацией)

// Просмотр списка и деталей выбранного контакта
void displayContacts() {

// Редактирование полей существующего контакта
void editContact() {
    if (!checkContactsExist()) return;

    printListOnly();

    printf("\nВведите номер контакта для редактирования (0 - отмена): ");
    int index = validateNumberInput(0, contactCount);
    if (index == 0) return;

    Item *current_item = getItemByIndex(index - 1);
    if (!current_item) return;

    // Сохраняем старый ключ - он нужен для удаления узла из дерева при сохранении
    Contact oldContact = current_item->contact;
    // Работаем с копией, чтобы не менять ключ в узле на месте и не ломать BST
    Contact tempContact = current_item->contact;

    char input[MAX_LENGTH];
    do {
        printContactDetailsByIndex(index);
        printf("\nРедактирование поля:\n");
        printf("1. Фамилия\n");
        printf("2. Имя\n");
        printf("3. Место работы\n");
        printf("4. Должность\n");
        printf("5. Телефон\n");
        printf("6. Email\n");
        printf("7. Соцсети\n");
        printf("8. Мессенджер\n");
        printf("9. Сохранить изменения\n");
        printf("0. Отменить изменения\n");
        printf("\nВыберите поле для редактирования: ");
        int field = validateNumberInput(0, 9);

        switch (field) {
            case 1:
                do {
                    printf("Новая фамилия: ");
                    getInput(input, MAX_LENGTH);
                } while (strlen(input) == 0);
                strcpy(tempContact.lastName, input);
                break;
            case 2:
                do {
                    printf("Новое имя: ");
                    getInput(input, MAX_LENGTH);
                } while (strlen(input) == 0);
                strcpy(tempContact.firstName, input);
                break;
            case 3:
                printf("Новое место: ");
                getInput(tempContact.workplace, MAX_LENGTH);
                break;
            case 4:
                printf("Новая должн: ");
                getInput(tempContact.job, MAX_LENGTH);
                break;
            case 5:
                printf("Новый телефон: ");
                getInput(tempContact.phone, MAX_LENGTH);
                break;
            case 6:
                printf("Новый email: ");
                getInput(tempContact.email, MAX_LENGTH);
                break;
            case 7:
                printf("Новые соцсети: ");
                getInput(tempContact.social, MAX_LENGTH);
                break;
            case 8:
                printf("Новый мессенджер: ");
                getInput(tempContact.messenger, MAX_LENGTH);
                break;
            case 9:
                // В BST нельзя менять ключ на месте - удаляем старый узел и вставляем новый
                deleteNode(oldContact, &root);
                insertNode(tempContact, &root);
                printf("\nИзменения сохранены!\n");
                waitEnter();
                return;
            case 0:
                printf("\nИзменения отменены.\n");
                waitEnter();
                return;
        }
    } while (1);
}

// Удаление контакта по индексу
void deleteContact() {
    if (!checkContactsExist()) return;

    printListOnly();

    printf("\nВведите номер контакта для удаления (0 - отмена): ");
    int index = validateNumberInput(0, contactCount);
    if (index == 0) return;

    Item *current_item = getItemByIndex(index - 1);
    if (!current_item) return;

    printf("\nВы уверены, что хотите удалить контакт #%d (%s %s)?\n",
           index, current_item->contact.lastName, current_item->contact.firstName);
    printf("1. Да, удалить\n");
    printf("2. Нет, отменить\n");
    printf("Выберите действие: ");

    if (validateNumberInput(1, 2) == 2) {
        printf("Удаление отменено.\n");
        waitEnter();
        return;
    }

    // Удаляем узел по значению ключа (фамилия + имя), а не по индексу
    deleteNode(current_item->contact, &root);
    contactCount--;
    operationsSinceBalance++;

    printf("\nКонтакт успешно удален!\n");

    // Периодическая балансировка после 5 операций
    if (operationsSinceBalance >= 5) balanceTree();

    waitEnter();
}
```

+ получение элементов по индексу,
